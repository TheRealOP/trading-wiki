import yfinance as yf
import pandas as pd
import numpy as np
import json
import sys
from datetime import datetime, timedelta
import warnings

warnings.filterwarnings('ignore')

def run_backtest(symbols, initial_capital, gap_threshold_pct, risk_per_trade_pct, stop_loss_pct):
    """
    Runs a backtest for the Mean Reversion Overnight Gap Fade strategy.
    """
    
    # -- Strategy Parameters --
    params = {
        "gap_threshold_pct": gap_threshold_pct,
        "risk_per_trade_pct": risk_per_trade_pct,
        "stop_loss_pct": stop_loss_pct
    }
    
    # -- Data Fetching --
    try:
        # yfinance period="2y" can be inconsistent, so we define the date range manually
        end_date = datetime.now()
        start_date = end_date - timedelta(days=2*365)
        data = yf.download(symbols, start=start_date, end=end_date, period="2y", auto_adjust=False)
        if data.empty:
            raise ValueError("No data downloaded from yfinance.")
    except Exception as e:
        # Handle case where yfinance fails
        print(json.dumps({
            "strategy": "Mean Reversion Overnight Gap Fade",
            "symbols": symbols,
            "period": f"{start_date.strftime('%Y-%m-%d')} to {end_date.strftime('%Y-%m-%d')}",
            "initial_capital": initial_capital,
            "error": f"Failed to download data: {str(e)}",
            "final_capital": 0, "total_return_pct": 0, "annualized_return_pct": 0,
            "sharpe_ratio": 0, "max_drawdown_pct": 0, "win_rate_pct": 0, "num_trades": 0,
            "params": params
        }), file=sys.stdout)
        return

    # -- Data Preparation --
    data['Prev_Close'] = data['Close'].shift(1)
    data['Gap_Pct'] = (data['Open'] - data['Prev_Close']) / data['Prev_Close'] * 100
    data.dropna(inplace=True)

    # -- Backtesting Loop --
    capital = initial_capital
    equity_curve = [initial_capital]
    trades = []
    
    for index, row in data.iterrows():
        gap = row['Gap_Pct']
        open_price = row['Open']
        close_price = row['Close']
        
        # Check for trading signal
        if abs(gap) > gap_threshold_pct:
            direction = 'Long' if gap < 0 else 'Short'
            
            # Position Sizing based on max risk per trade
            capital_at_risk = capital * (risk_per_trade_pct / 100.0)
            risk_per_share = open_price * (stop_loss_pct / 100.0)
            
            if risk_per_share == 0:
                continue

            num_shares = int(capital_at_risk / risk_per_share)
            
            if num_shares == 0:
                continue # Can't afford even one share with the risk constraint

            position_value = num_shares * open_price
            
            if position_value > capital:
                # Recalculate shares based on available capital if risk sizing is too large
                num_shares = int(capital / open_price)
                if num_shares == 0:
                    continue
            
            # Execute trade and calculate PnL
            if direction == 'Long':
                pnl = (close_price - open_price) * num_shares
            else: # Short
                pnl = (open_price - close_price) * num_shares
                
            capital += pnl
            equity_curve.append(capital)
            
            trades.append({
                "entry_date": index.strftime('%Y-%m-%d'),
                "direction": direction,
                "shares": num_shares,
                "entry_price": open_price,
                "exit_price": close_price,
                "pnl": pnl
            })
            
    # -- Performance Metrics Calculation --
    num_trades = len(trades)
    
    if num_trades == 0:
        final_capital = initial_capital
        total_return_pct = 0.0
        annualized_return_pct = 0.0
        sharpe_ratio = 0.0
        max_drawdown_pct = 0.0
        win_rate_pct = 0.0
    else:
        final_capital = capital
        total_return_pct = ((final_capital / initial_capital) - 1) * 100
        
        # Annualized Return
        total_days = (data.index[-1] - data.index[0]).days
        years = max(total_days / 365.25, 1) # Avoid division by zero for short periods
        annualized_return_pct = ((final_capital / initial_capital) ** (1 / years) - 1) * 100 if years > 0 else 0

        # Sharpe Ratio
        equity_df = pd.DataFrame(equity_curve, columns=['Equity'])
        daily_returns = equity_df['Equity'].pct_change().dropna()
        if daily_returns.std() > 0:
            sharpe_ratio = (daily_returns.mean() / daily_returns.std()) * np.sqrt(252)
        else:
            sharpe_ratio = 0.0
            
        # Max Drawdown
        rolling_max = equity_df['Equity'].cummax()
        drawdown = (equity_df['Equity'] / rolling_max) - 1
        max_drawdown_pct = drawdown.min() * 100
        
        # Win Rate
        wins = sum(1 for trade in trades if trade['pnl'] > 0)
        win_rate_pct = (wins / num_trades) * 100

    # -- Final JSON Output --
    period_str = f"{data.index[0].strftime('%Y-%m-%d')} to {data.index[-1].strftime('%Y-%m-%d')}"
    
    results = {
        "strategy": "Mean Reversion Overnight Gap Fade",
        "symbols": symbols,
        "period": period_str,
        "initial_capital": initial_capital,
        "final_capital": round(final_capital, 2),
        "total_return_pct": round(total_return_pct, 2),
        "annualized_return_pct": round(annualized_return_pct, 2),
        "sharpe_ratio": round(sharpe_ratio, 2),
        "max_drawdown_pct": round(max_drawdown_pct, 2),
        "win_rate_pct": round(win_rate_pct, 2),
        "num_trades": num_trades,
        "params": params
    }
    
    print(json.dumps(results, indent=4), file=sys.stdout)

if __name__ == '__main__':
    # -- Configuration --
    SYMBOLS = ["SPY"]
    INITIAL_CAPITAL = 500
    GAP_THRESHOLD_PCT = 1.0  # e.g., 1.0 for 1%
    RISK_PER_TRADE_PCT = 2.0 # Max % of portfolio to risk on a single trade
    STOP_LOSS_PCT = 0.5      # % below/above entry for stop loss, used for sizing
    
    run_backtest(
        symbols=SYMBOLS,
        initial_capital=INITIAL_CAPITAL,
        gap_threshold_pct=GAP_THRESHOLD_PCT,
        risk_per_trade_pct=RISK_PER_TRADE_PCT,
        stop_loss_pct=STOP_LOSS_PCT
    )
